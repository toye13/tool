// ==UserScript== // @name Discord ボイチャ酸化機能（復活祭ネタ） // @namespace http://tampermonkey.net/ // @version 2025.2.11-rust // @description 話すたびに声が錆びていく…完全酸化まであと何分？ // @author パソコン専門 // @match https://discord.com/channels/* // @grant none // ==/UserScript== (function() { 'use strict'; let oxidationLevel = 0; // 0〜100（100で完全酸化） let lastSpeakTime = Date.now(); let isSpeaking = false; let audioContext = null; let voiceStream = null; // 酸化エフェクト適用関数 function applyOxidation(stream) { if (!audioContext) { audioContext = new (window.AudioContext || window.webkitAudioContext)(); } const source = audioContext.createMediaStreamSource(stream); const gain = audioContext.createGain(); const biquad = audioContext.createBiquadFilter(); const distortion = audioContext.createWaveShaper(); // 酸化レベルに応じてエフェクト強度変更 gain.gain.value = 1 - (oxidationLevel / 200); // 徐々に音量ダウン biquad.type = 'lowshelf'; biquad.frequency.value = 800 - (oxidationLevel * 6); biquad.gain.value = -oxidationLevel / 2; // ガリガリノイズ（錆び感） distortion.curve = new Float32Array(65536).map((_, i) => { const x = (i - 32768) / 32768; return x * (oxidationLevel / 30) * Math.sin(x * 10); }); source.connect(gain); gain.connect(biquad); biquad.connect(distortion); distortion.connect(audioContext.destination); return stream; } // 話してるか監視（簡易版） function monitorSpeaking() { setInterval(() => { const now = Date.now(); if (isSpeaking) { // 話してる間は酸化進行 oxidationLevel = Math.min(100, oxidationLevel + 0.3); lastSpeakTime = now; } else if (now - lastSpeakTime > 5000) { // 5秒以上喋ってないと少し回復 oxidationLevel = Math.max(0, oxidationLevel - 1); } // 酸化レベル表示（オーバーレイ） let overlay = document.getElementById('oxidation-overlay'); if (!overlay) { overlay = document.createElement('div'); overlay.id = 'oxidation-overlay'; overlay.style.position = 'fixed'; overlay.style.top = '10px'; overlay.style.right = '10px'; overlay.style.background = 'rgba(0,0,0,0.7)'; overlay.style.padding = '8px 16px'; overlay.style.borderRadius = '8px'; overlay.style.zIndex = '99999'; overlay.style.fontSize = '1.2em'; overlay.style.pointerEvents = 'none'; document.body.appendChild(overlay); } overlay.textContent = `酸化度: ${Math.floor(oxidationLevel)}% ${oxidationLevel > 80 ? '🦴完全に錆びてる…' : oxidationLevel > 50 ? '🧲錆び始めてる…' : ''}`; }, 1000); } // ボイスチャンネル参加を検知してフック const originalGetUserMedia = navigator.mediaDevices.getUserMedia; navigator.mediaDevices.getUserMedia = async function(constraints) { const stream = await originalGetUserMedia.call(navigator.mediaDevices, constraints); // マイクトラックが見つかったらエフェクト適用 const audioTrack = stream.getAudioTracks()[0]; if (audioTrack) { console.log('[酸化機能] マイク検知 → 錆び開始'); isSpeaking = true; return applyOxidation(stream); } return stream; }; // 話してるか監視開始 monitorSpeaking(); // チャンネル退出検知（簡易） setInterval(() => { if (!document.querySelector('[aria-label*="ボイスチャンネル"]')) { oxidationLevel = 0; // 退出したらリセット isSpeaking = false; } }, 2000); console.log('【復活祭ネタ】ボイチャ酸化機能 起動！ 話せば話すほど錆びていきます…🦴'); })();
